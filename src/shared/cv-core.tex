\newcommand*{\cv@makeheader}{
    \ifx\@hasfoto\undefined
        % No photo provided: centered layout
        \begin{center}
            \gebruiker{\@voornaam}{\@achternaam}
            \cvkop{\@tagline}
            {\small{\@socialeinfo}}
        \end{center}
    \else
        % Photo provided: split layout
        \begin{minipage}{\dimexpr\linewidth-\@fotodiameter-2em}
            \gebruiker{\@voornaam}{\@achternaam}
            \cvkop{\@tagline}
            {\small{\@socialeinfo}}
        \end{minipage}
        \begin{minipage}{\@fotodiameter}
            \idfoto
        \end{minipage}
    \fi
    \bigskip
}
